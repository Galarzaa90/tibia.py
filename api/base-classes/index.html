def __eq__(self, o: object) -> bool:

def __eq__(self, o: object) -> bool:
    """Two houses are considered equal if their names or ids are equal."""
    if isinstance(o, self.__class__):
        return self.name.lower() == o.name.lower() or self.id == o.id

    return False
